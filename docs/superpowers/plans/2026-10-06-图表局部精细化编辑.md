```python
# 后端_核心/refine_report.py
"""报表图表局部精细化编辑服务（阶段 55）。
只做图表层局部修改：不触发 Agent ReAct、不调用外部工具/LLM，
仅复用报表关联数据集做本地 pandas 重算，另存新报表。
"""
import re
from typing import Dict, List


_标识符模式 = re.compile(r"(id|编号|编码|code|序号|单号|号码|uuid|标识|键)$", re.I)


def 标识符字段(画像: Dict) -> List[str]:
    """识别标识符/编码 ID 字段（禁区：禁作 Y 轴/数值度量，可作 X 轴分类）。

    识别规则：
    1. 字段名命中 id/编号/编码/code/序号/单号/uuid/标识 等模式；
    2. 高基数数值字段（唯一值数 ≈ 行数，如 user_id/score 类 ID）。
    """
    行数 = int(画像.get("行数", 0)) or 1
    out = []
    # 画像需含唯一值数——用字段名模式 + 数值高基数启发即可，画像不显式存唯一数
    for field in 画像.get("字段列表", []):
        if _ID模式.search(str(field)):
            out.append(field)
    return list(dict.fromkeys(out))  # 去重保序
```

```python
# 后端_核心/refine_report.py 追加
from 后端_核心.field_selector import _合法字段

_允许图表类型 = {"柱状图", "折线图", "饼图", "散点图", "面积图", "直方图"}

_数值动作 = {"改标题", "改子标题", "改图例名称", "改备注"}
_轴动作 = {"换X轴", "换Y轴", "改X轴别名", "改Y轴别名", "改坐标轴范围"}
_切换动作 = {"切图表类型"}


def _换标识符禁区(画像, 图表类型):
    # Y 轴禁止：标识符/编码；X 轴允许
    return 标识符字段(画像)


def 编辑校验(编辑, 画像):
    """校验编辑动作 - 返回 (ok, 拒绝原因)。异常返回 (False, 原因)。"""
    动作 = 编辑.get("动作")
    if not 动作:
        return False, "缺少编辑动作"
    if 动作 in _数值动作:
        return True, None
    if 动作 in _轴动作:
        # X 轴允许标识符、Y 轴禁止
        return True, None
    if 动作 == "换Y轴":
        y = 编辑.get("Y轴", [])
        for f in y:
            if f in _校验标识符禁区(画像, f):
                return False, f"字段「{f}」是标识符/编码，禁止作为数值度量 Y 轴，仅能作分类 X 轴"
        return True, None
    if 动作 == "换X轴":
        x = 编辑.get("X轴")
        if x is not None and x not in 画像.get("字段列表", []):
            return False, f"字段「{x}」不存在于数据集中"
        return True, None
    if 动作 in _切换动作:
        return True, None
    return True, None


def 应用编辑(chart_config, 编辑):
    """在既有 chart_config 上做纯 spec 层面的修改（不改 数据，仅改展示字段）。"""
    new = dict(chart_config)
    动作 = 编辑.get("动作")
    if 动作 == "改标题":
        new["标题"] = 编辑["标题"]
    elif 动作 == "改子标题":
        new["子标题"] = 编辑.get("子标题")
    elif 动作 == "改图例名称":
        new["图例"] = 编辑.get("图例")
    elif 动作 == "改备注":
        new["备注"] = 编辑.get("备注")
    elif 动作 == "换X轴":
        new["X轴"] = 编辑["X轴"]
    elif 动作 == "换Y轴":
        new["Y轴"] = 编辑["Y轴"]
    elif 动作 == "改X轴别名":
        new["X轴别名"] = 编辑.get("别名")
    elif 动作 == "改Y轴别名":
        new["Y轴别名"] = 编辑.get("别名")
    elif 动作 == "改坐标范围":
        new["坐标轴范围"] = 编辑.get("范围")
    elif 动作 == "切图表类型":
        new["类型"] = 编辑["图表类型"]  # 数据重算在 Task3 做；此处仅切渲染类型便签
    return new
```

```python
_标题模式 = re.compile(r"(标题|题目)[改叫为叫改为改成]?.{0,4}?：?(.+)")  # 需更细

    # 用 field_selector._匹配意图字段 或 直接命中
    from 后端_核心.field_selector import _匹配意图字段
    关键词 = [候选]
    return _匹配意图字段(画像, 关键词)

def 解析编辑指令(文本, 画像):
    文本 = 文本.strip()
    # 标题
    m = re.search(r"标题改?为?[：:]?(.+[^。.;；])", 文本)
    if m and "轴" not in 文本:
        return {"动作": "改标题", "标题": m.group(1).strip()}, False, None
    # X 轴
    m = re.search(r"[Xx]\s*轴(?:换|改为|改成)?[：:]?\s*([^\s,，。]+)", 文本)
    if m:
        f = 命中字段(m.group(1), 画像)
        if f: return {"动作": "换X轴", "X轴": f}, False, None
        return None, True, f"X轴字段「{m.group(1)}」未识别，请确认"
    # Y 轴
    m = re.search(r"[Yy]\s*轴(?:换|改为|改成)?[：:]?\s*([^\s,，。]+)", 文本)
    if m:
        f = 命中字段(m.group(1), 画像)
        if f: return {"动作": "换Y轴", "Y轴": [f]}, False, None
        return None, True, f"Y轴字段「{m.group(1)}」未识别，请确认"
    # 切图表类型
    for 原, 目标 in _词义词.items():
        if 原 in 文本 or 目标 in 文本:
            if 目标 in _允许图表类型:
```

```python
from 后端_核心.report_generator import _聚合数据, _可_json行

_占比模式 = {"计数": "count", "数值": "sum"}

def 重算图表数据(画像, df, 编辑, 原_chart):
    动作 = 编辑.get("动作")
    新 = dict(原_chart)
    x = 新.get("X轴"); y = 新.get("Y轴", [])
    agg = "计数" if not y else "求和"
    if 动作 == "换X轴":
        x = 编辑["X轴"]
    elif 动作 == "换Y轴":
        y = 编辑["Y轴"]
    # 加筛选：对原始 df 先行过滤
    if 动作 == "加筛选":
        cond = 编辑.get("筛选")
        df = df[df[cond["字段"]] == cond["值"]]
    # 字段仅在原数据列里重算
    avail = list(df.columns)
    if x not in avail: x = avail[0] if avail else None
    y = [yy for yy in y if yy in avail]
    agg_method = _占比模式.get(编辑.get("占比"), "sum") if 编辑.get("动作") == "切占比" else agg
    报告df = _聚合数据(df, x, y or ["记录数"], None, agg_method)
    新["数据"] = _可_json行(报告df)
    新["X轴"] = x
    新["Y轴"] = y or ["记录数"]
    新["占比模式"] = "计数" if agg_method == "count" else "数值"
    注明 = f"已按 {x} 重算（{agg_method}）"
    return 新, 说明
```

```python
@router.post("/{report_id}/refine")
def refine_report_route(report_id: str, body: dict, user: dict = Depends(get_current_user)):
    from 后端_核心 import refine_report as rr
    报表 = report_repo.读取报表(user["user_id"], report_id)
    if not 报表:
        raise HTTPException(status_code=404, detail="报表不存在")
    ds = 读取数据集(user["user_id"], 报表["数据集ID"])
    画像 = ds["数据画像"]; df = ds["数据"]
    原配置 = 报表["报表"].get("图表配置") or {}
    指令 = body.get("指令")
    if 指令:
        edit, need, err = rr.解析编辑指令(指令, 画像)
    else:
        edit, need, err = body.get("编辑"), False, None
    if need:
        return {"需确认": True, "候选": [], "拒绝原因": err}
    ok, reason = rr.编辑校验(edit, 画像)
    if not ok:
        return {"需确认": False, "新报表ID": None, "新spec": None, "拒绝原因": reason}
    新配置, 说明 = rr.重算图表数据(画像, df, edit, 原配置) if 需要重算 else rr.应用编辑(原配置, edit)
    变更清单 = 生成变更清单(原配置, 新配置)
    新report = dict(报表["报表"]); 新report["图表配置"] = 新配置
    新report["来源报表ID"] = report_id; 新report["标题"] = 新配置.get("标题")
    neue_id = report_repo.保存报表(user["user_id"], 报表["数据集ID"],
                                   新配置.get("标题") or 报表["标题"],
                                   新配置.get("类型") or 报表["图表类型"], 新report)
    return {"新报表ID": 新report_id, "新spec": 新配置, "变更清单": 变更清单,
            "需确认": False, "拒绝原因": None}
```
