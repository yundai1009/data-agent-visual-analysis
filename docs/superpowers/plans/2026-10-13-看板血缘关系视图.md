## 2. 设计决策（含理由）

1. **不套"神经网络皮肤"，只借它的图谱规则。**
   `neural-creator-dashboard/src/neural/graph.js` 有一条规则：**"同一概念被 ≥2 条内容提及才成为概念节点"**。
   等价映射到本项目：**"同一字段被 ≥2 张报表引用才成为字段节点"**——这就是"高价值字段"的定义（字段复用榜）。
   被 1 张报表引用的字段不建节点（进抽屉的报表明细里），避免图被一次性字段撑爆。

2. **不画"看板"节点。**
   落点里写的是"数据集 → 报表 → 看板"，但当前页**就是**这块看板，图里所有报表天然都属于它，
   再画一个看板节点 + N 条边只增加噪声，还和右侧字段层抢终端位置。
   改为：看板身份用**图上方标题条**（`当前看板：{名称} · X 个数据集 → Y 张报表 → Z 个复用字段`）表达。

3. **字段按 `数据集ID::字段名` 归并，不按字段名全局归并。**
   两个数据集都叫「金额」是**两个不同的字段**，混成一个节点会得出"字段被 3 张报表复用"的假结论。

4. **反链（跨看板发现）挂在抽屉，不画成节点。**
   同一数据集下没进本看板的历史报表（来自 `listReports` 全量）列在**数据集节点的抽屉**里。
   这样既拿到了"这份数据还关联了什么"，又不把图撑成第四层。

5. **手动布局（`layout: 'none'` + 归一化坐标）而不是力导向。**
   三层血缘的 x 是确定的（左/中/右），y 均匀铺开 → 坐标在纯函数里算好，
   好处是 **option 可被单测逐字段断言**；力导向每次渲染位置都抖，测不了、截图也不稳定。

6. **新组件独立，不塞进 `buildOption`。**
   `EChartsChart.buildOption` 是给 18 种业务图表用的（输入 `图表配置.数据` 行数组），
   关系图输入是图结构 + 点击交互，硬塞进去会污染已有 498 行 + 已有回归测试。
   但 `LineageGraph` 的**生命周期写法**（懒加载 echarts / `notMerge` / `dispose` / resize 监听）与 `EChartsChart` 保持一致，
   避免仓库里出现两套图表生命周期。

7. **色板/字体/HTML 转义抽到 `utils/图表主题.js` 单一来源。**
   血缘节点颜色必须和报表图是同一套藏青系，否则同一页两种蓝。顺带让 P0-6 的 `escapeHtml` 只有一个实现。

/** 藏青系协调色板（阶段 13 重设计：与 UI 主色统一） */
export const COLORS = ['#0f4c81', '#3d7bb8', '#5b8cb8', '#8aa9c4', '#b45309', '#0f766e', '#64748b', '#334155'];

export const CHART_TOOLTIP = {

  borderColor: '#e2e8f0',
  borderWidth: 1,
  textStyle: { fontFamily: 'Noto Sans SC, Microsoft YaHei, sans-serif', fontSize: 12, color: '#1e293b' },
  extraCssText: 'box-shadow: 0 8px 16px -8px rgba(15,76,129,.15); border-radius: 8px;',
};

/**
 * HTML 转义（P0 加固，审计实测 P0-6）。
 * echarts tooltip 内容按 HTML 渲染，自定义 formatter 返回值必须自行转义，
 * 否则上传数据里的 <img onerror=...> 可触发存储型 XSS。字段名/文件名都来自用户上传数据，必转。
 */
export function 转义HTML(value) {
  return String(value ?? '').replace(/[&<>"']/g, (c) => (
    { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]
  ));
}
```

/**
 * 阶段 六十 · 看板血缘关系视图：纯函数层。
 *
 * 做了什么：把三个既有接口的响应拼成一张有向图
 *          数据集(左) → 报表(中) → 字段(右)，供 ECharts graph 渲染。
 * 为什么这样设计：
 *   - 看板只显示"这几张图并排"，看不出这些图共用哪份数据、哪几列；
 *     血缘图回答"这个数字是从哪来、还被谁用了"，是分析的溯源需求；
 *   - 字段层规则借鉴开源项目 neural-creator-dashboard 的图谱思路
 *     （PolyForm Noncommercial 许可，只借规则不抄代码）：
 *     "同一概念被 ≥2 条内容提及才成为概念节点"
 *     → 等价为 "同一字段被 ≥2 张报表引用才成为字段节点"，即高价值字段。
 *
 * 纯函数、无副作用、数据缺失一律降级不抛异常（可直接单测）。
 */

import { COLORS, CHART_TOOLTIP, 转义HTML } from './图表主题';

const 未知数据集ID = '__未知数据集__';
const 复用阈值 = 2;          // 被 ≥2 张报表引用才算高价值字段

/** 抽字段并记角色；只认图表配置里真正出现过的列名（挡掉 boxplot 的 name/value 哨兵） */
export function 提取图表字段(图表配置) {
  const cfg = 图表配置 || {};
  const 数据键 = new Set(
    Array.isArray(cfg.数据) && cfg.数据[0] && typeof cfg.数据[0] === 'object'
      ? Object.keys(cfg.数据[0])
      : [],
  );
  const 需校验 = 数据键.size > 0;   // 老报表没有数据行时跳过校验，宁可多列也不要凭空捏造

  const 角色表 = new Map();
  const 记 = (候选, 角色) => {
    if (typeof 候选 !== 'string') return;
    const 名 = 候选.trim();
    if (!名) return;
    if (需校验 && !数据键.has(名)) return;
    if (!角色表.has(名)) 角色表.set(名, new Set());
    角色表.get(名).add(角色);
  };

  记(cfg.X轴, 'X轴');
  (Array.isArray(cfg.Y轴) ? cfg.Y轴 : []).forEach((f) => 记(f, 'Y轴'));
  记(cfg.颜色 ?? cfg.分组字段, '分组');
  记(cfg.名称, '名称');
  记(cfg.值, '数值');

  return [...角色表.entries()].map(([字段名, 角色集]) => ({ 字段名, 角色: [...角色集] }));
}

/** 结论摘要：折空白 + 截断（抽屉里一行放得下） */
function 结论摘要(文本, 上限 = 80) {
  const s = String(文本 ?? '').replace(/\s+/g, ' ').trim();
  return s.length > 上限 ? `${s.slice(0, 上限)}…` : s;
}

/**
 * 构建血缘图。
 * @param {Array} 入参.报表列表   getDashboard(id).报表列表（不含数据集ID）
 * @param {Map}   入参.报表索引   报表ID → listReports 列表项（含数据集ID）
 * @param {Map}   入参.数据集索引 数据集ID → listDatasets 列表项（含文件名/行数/列数）
 * @returns {{节点: Array, 连线: Array, 节点索引: Object, 统计: Object}}
 */
export function 构建血缘图({ 报表列表 = [], 报表索引 = new Map(), 数据集索引 = new Map() } = {}) {
  const 节点 = [];
  const 连线 = [];
  const 数据集节点表 = new Map();   // 数据集ID → 节点
  const 报表节点表 = new Map();     // 报表ID → 节点
  const 字段累计 = new Map();       // `数据集ID::字段名` → { 字段名, 数据集ID, 角色:Set, 报表:[] }

  (Array.isArray(报表列表) ? 报表列表 : []).forEach((项) => {
    const 报表ID = 项?.报表ID;
    if (!报表ID) return;
    const 元 = 报表索引.get(报表ID) || {};
    const 数据集ID = 元.数据集ID || 未知数据集ID;   // listReports 截断/旧报表 → 未知数据集
    const 标题 = 项.标题 || 元.标题 || '未命名报表';
    const 图表类型 = 项.图表类型 || 元.图表类型 || '';

    // 层 1：数据集节点（同一数据集只建一次）
    if (!数据集节点表.has(数据集ID)) {
      const ds = 数据集索引.get(数据集ID) || {};
      const 节点 = {
        节点ID: `ds:${数据集ID}`,
        名称: 数据集ID === 未知数据集ID ? '未知数据集' : (ds.文件名 || 数据集ID),
        类型: '数据集',
        层: 0,
        数据集ID,
        度: 0,
        明细: {
          文件名: ds.文件名 || '',
          行数: ds.行数 ?? null,
          列数: ds.列数 ?? null,
          来源数据集ID: ds.来源数据集ID || '',
          本看板报表: [],
          其他报表: [],   // 同数据集但没进本看板的历史报表（反链发现）
        },
      };
      数据集节点表.set(数据集ID, 节点);
      节点.push(节点);
    }
    const 数据集节点 = 数据集节点表.get(数据集ID);

    // 层 2：报表节点
    const 字段明细 = 提取图表字段(项?.报表?.图表配置);
    const 报表节点 = {
      节点ID: `rp:${报表ID}`,
      名称: 标题,
      类型: '报表',
      层: 1,
      报表ID,
      数据集ID,
      图表类型,
      度: 字段明细.length + 1,
      明细: {
        标题,
        图表类型,
        创建时间: 项.创建时间 || 元.创建时间 || '',
        数据集ID,
        数据集名: 数据集节点.名称,
        结论: 结论摘要(项?.报表?.结论),
        字段: 字段明细,
      },
    };
    if (!报表节点表.has(报表ID)) {
      报表节点表.set(报表ID, 报表节点);
      节点.push(报表节点);
    }

    数据集节点.明细.本看板报表.push({ 报表ID, 标题, 图表类型 });
    数据集节点.度 += 1;
    连线.push({
      连线ID: `ds:${数据集ID}->rp:${报表ID}`,
      起点: 数据集节点.节点ID,
      终点: 报表节点.节点ID,
      类型: '归属',
    });

    // 字段累计：按 数据集::字段 归并（同名不同数据集不合并，否则会算出假的复用度）
    字段明细.forEach(({ 字段名, 角色 }) => {
      const 键 = `${数据集ID}::${字段名}`;
      if (!字段累计.has(键)) 字段累计.set(键, { 字段名, 数据集ID, 角色: new Set(), 报表: [] });
      const 累计 = 字段累计.get(键);
      角色.forEach((r) => 累计.角色.add(r));
      累计.报表.push({ 报表ID, 标题, 图表类型 });
    });
  });

  // 层 3：只收被 ≥阈值 张报表引用的字段（高价值字段）；其余字段留在报表节点明细里
  const 高价值字段 = [];
  [...字段累计.entries()]
    .sort((a, b) => b[1].报表.length - a[1].报表.length || a[1].字段名.localeCompare(b[1].字段名, 'zh-CN'))
    .forEach(([键, 累计]) => {
      if (累计.报表.length < 复用阈值) return;
      const 数据集节点 = 数据集节点表.get(累计.数据集ID);
      const 字段节点 = {
        节点ID: `fd:${键}`,
        名称: 累计.字段名,
        类型: '字段',
        层: 2,
        字段名: 累计.字段名,
        数据集ID: 累计.数据集ID,
        度: 累计.报表.length,
        明细: {
          字段名: 累计.字段名,
          数据集名: 数据集节点?.名称 || '',
          角色: [...累计.角色],
          报表: 累计.报表,
        },
      };
      节点.push(字段节点);
      高价值字段.push(字段节点);
      累计.报表.forEach((r) => {
        连线.push({
          连线ID: `rp:${r.报表ID}->fd:${键}`,
          起点: `rp:${r.报表ID}`,
          终点: 字段节点.节点ID,
          类型: '引用',
        });
      });
    });

  // 反链：同数据集、但没进本看板的历史报表
  数据集节点表.forEach((数据集节点) => {
    const 已有 = new Set(数据集节点.明细.本看板报表.map((r) => r.报表ID));
    数据集节点.明细.其他报表 = [...报表索引.values()]
      .filter((r) => r?.数据集ID === 数据集节点.数据集ID && !已有.has(r.报表ID))
      .map((r) => ({ 报表ID: r.报表ID, 标题: r.标题, 图表类型: r.图表类型 }));
  });

  return {
    节点,
    连线,
    节点索引: Object.fromEntries(节点.map((n) => [n.节点ID, n])),
    统计: {
      数据集数: 数据集节点表.size,
      报表数: 报表节点表.size,
      字段总数: 字段累计.size,
      高价值字段: 高价值字段.map((n) => ({
        节点ID: n.节点ID,
        字段名: n.字段名,
        数据集名: n.明细.数据集名,
        报表数: n.度,
        角色: n.明细.角色,
        报表: n.明细.报表,
      })),
    },
  };
}

// ---- option 生成（也是纯函数，单测直接断言结构）----

const 层X = { 0: 16, 1: 50, 2: 84 };   // 归一化 0-100 坐标，ECharts graph layout:'none' 会自适应视口
const 类别索引 = { 数据集: 0, 报表: 1, 字段: 2 };
const 基础底色 = { 数据集: COLORS[0], 报表: COLORS[1], 字段: COLORS[5] };

/**
 * 生成关系图 option。
 * @param {Object} 图数据 构建血缘图 的返回值
 * @param {Object} 配置   高亮节点ID（点击后持久高亮自己算，ECharts 的 emphasis.focus 只管 hover）
 */
export function 构建血缘Option(图数据, { 高亮节点ID = '' } = {}) {
  const 节点们 = 图数据?.节点 || [];
  const 连线们 = 图数据?.连线 || [];
  const 索引 = 图数据?.节点索引 || {};

  // 分层铺 y：同层均匀分布，避免节点重叠在同一条线上
  const 分层 = { 0: [], 1: [], 2: [] };
  节点们.forEach((n) => { (分层[n.层] || 分层[1]).push(n); });
  const 坐标 = new Map();
  Object.entries(分层).forEach(([层, 组]) => {
    组.forEach((n, i) => 坐标.set(n.节点ID, { x: 层X[层], y: (100 * (i + 1)) / (组.length + 1) }));
  });

  // 邻接集合：选中节点自身 + 上下游
  const 邻接 = new Set();
  if (高亮节点ID) {
    邻接.add(高亮节点ID);
    连线们.forEach((l) => {
      if (l.起点 === 高亮节点ID) 邻接.add(l.终点);
      if (l.终点 === 高亮节点ID) 邻接.add(l.起点);
    });
  }
  const 淡 = (id) => (高亮节点ID && !邻接.has(id) ? 0.18 : 1);

  return {
    animationDuration: 600,
    legend: {
      data: ['数据集', '报表', '字段（≥2 张报表复用）'],
      top: 0,
      left: 0,
      itemWidth: 10,
      itemHeight: 10,
      textStyle: { fontSize: 11, color: '#64748b' },
    },
    tooltip: {
      ...CHART_TOOLTIP,
      trigger: 'item',
      confine: true,   // 图在页面中间，弹层不越出容器
      formatter: (p) => {
        if (p?.dataType === 'edge') return 转义HTML(p.data?.提示 || '引用');
        const n = p?.data?.节点ID ? 索引[p.data.节点ID] : null;
        if (!n) return 转义HTML(p?.name || '');
        const 行 = [`<b>${转义HTML(n.名称)}</b>`, `类型：${n.类型}`];
        if (n.类型 === '数据集') 行.push(`本看板引用 ${n.度} 张报表`);
        if (n.类型 === '报表') 行.push(`图表：${转义HTML(n.图表类型 || '未知')}`);
        if (n.类型 === '字段') 行.push(`被 ${n.度} 张报表引用`);
        行.push('<span style="color:#94a3b8">单击查看详情</span>');
        return 行.join('<br/>');
      },
    },
    series: [{
      type: 'graph',
      layout: 'none',        // 坐标在上面算好：确定性 → 可单测、截图稳定
      roam: true,            // 滚轮缩放 + 拖拽平移
      draggable: true,
      symbol: 'roundRect',
      edgeSymbol: ['none', 'arrow'],
      edgeSymbolSize: [0, 6],
      label: { show: true, position: 'bottom', distance: 4, fontSize: 11, color: '#475569' },
      lineStyle: { color: '#cbd5e1', width: 1.2, curveness: 0.08 },
      emphasis: { focus: 'adjacency', lineStyle: { width: 2.4 } },
      categories: [
        { name: '数据集', itemStyle: { color: 基础底色.数据集 } },
        { name: '报表', itemStyle: { color: 基础底色.报表 } },
        { name: '字段（≥2 张报表复用）', itemStyle: { color: 基础底色.字段 } },
      ],
      data: 节点们.map((n) => {
        const 底色 = n.类型 === '字段' && n.度 >= 3 ? COLORS[4] : 基础底色[n.类型];  // 3 张以上复用 → 琥珀色警示
        const 基数 = n.类型 === '数据集' ? [26, 30] : n.类型 === '报表' ? [22, 26] : [18, 22];
        const 增量 = n.类型 === '字段' ? Math.min(n.度, 5) * 3 : 0;   // 复用度越高字段越大
        return {
          id: n.节点ID,
          节点ID: n.节点ID,
          name: n.名称,
          category: 类别索引[n.类型] ?? 1,
          x: 坐标.get(n.节点ID)?.x ?? 50,
          y: 坐标.get(n.节点ID)?.y ?? 50,
          symbolSize: [基数[0] + 增量, 基数[1] + Math.round(增量 * 0.7)],
          itemStyle: { color: 底色, opacity: 淡(n.节点ID), borderColor: '#ffffff', borderWidth: 1.5 },
          label: { opacity: 淡(n.节点ID) },
        };
      }),
      links: 连线们.map((l) => ({
        source: l.起点,
        target: l.终点,
        提示: l.类型,
        lineStyle: { type: l.类型 === '引用' ? 'dashed' : 'solid', opacity: 淡(l.起点) * 淡(l.终点) },
      })),
    }],
  };
}

// 阶段 六十 · 看板血缘关系视图的 ECharts 渲染层。
//
// 做了什么：把 utils/看板血缘.js 算好的 节点/连线 画成有向关系图，并把点击的节点ID回传父组件。
// 为什么单独一个组件、不复用 EChartsChart：
//   - EChartsChart 的输入是"报表数据 + 图表配置"（buildOption 面向 18 种业务图表），
//     关系图的输入是图结构 + 点击交互，硬塞进去会污染那条已测路径；
//   - 但渲染范式（懒加载 echarts / notMerge / dispose / resize）保持一致，
//     避免仓库里出现两套图表生命周期写法。
import { useEffect, useRef, useState } from 'react';
import { 构建血缘Option } from '../utils/看板血缘';

export default function LineageGraph({ 图数据, 高亮节点ID = '', onSelectNode, height = 520 }) {
  const 容器 = useRef(null);
  const 图实例 = useRef(null);
  const echarts模块 = useRef(null);
  const 回调 = useRef(onSelectNode);      // 事件回调用 ref 取，避免注册时闭包过期
  const [就绪, set就绪] = useState(false);

  useEffect(() => { 回调.current = onSelectNode; }, [onSelectNode]);

  // echarts 懒加载：与 EChartsChart 同款（约 1MB 独立 chunk，不进主 bundle）
  useEffect(() => {
    let 已取消 = false;
    (async () => {
      try {
        const mod = await import('echarts');
        if (已取消) return;
        echarts模块.current = mod;
        set就绪(true);
      } catch (e) {
        console.error('图表库加载失败:', e);
      }
    })();
    return () => { 已取消 = true; };
  }, []);

  useEffect(() => {
    if (!就绪 || !容器.current) return;
    const echarts = echarts模块.current;
    if (!echarts) return;
    if (!图实例.current) 图实例.current = echarts.init(容器.current, null, { renderer: 'canvas' });
    const chart = 图实例.current;

    try {
      const option = 构建血缘Option(图数据, { 高亮节点ID });
      if (option) {
        chart.setOption(option, { notMerge: true });
        chart.resize();
      }
    } catch (e) {
      console.error('血缘图渲染异常:', e);
      chart.setOption({
        title: { text: `关系图渲染异常: ${e.message}`, left: 'center', textStyle: { fontSize: 13, color: '#ef4444' } },
      }, { notMerge: true });
    }

    // 点击节点回传（实例每次都重建，dispose 后监听随之失效，无需手动 off）
    chart.on('click', (p) => {
      if (p?.dataType !== 'node') return;
      const id = p.data?.节点ID;
      if (id) 回调.current?.(id);
    });

    return () => {
      if (图实例.current) {
        图实例.current.dispose();
        图实例.current = null;
      }
    };
    // 图数据每次渲染都是新对象引用 → 用内容签名，避免 dispose+重建循环（同 EChartsChart）
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [就绪, 高亮节点ID, JSON.stringify(图数据?.节点), JSON.stringify(图数据?.连线)]);

  useEffect(() => {
    const handler = () => { if (图实例.current) 图实例.current.resize(); };
    window.addEventListener('resize', handler);
    return () => window.removeEventListener('resize', handler);
  }, []);

  return (
    <div className="relative w-full" style={{ height }}>
      {!就绪 && <p className="absolute inset-0 flex items-center justify-center text-xs text-gray-400">关系图加载中…</p>}
      <div ref={容器} className="w-full h-full" data-testid="血缘图画布" />
    </div>
  );
}
```

## Task 5 · 可选加固：报表超出 `listReports(100)` 时补取

**为什么需要**：`listReports` 的 `limit` 上限是 100（`api/routes/reports.py:597` `limit: int = Query(50, ge=1, le=100)`），
而 `Dashboard.jsx:81` 取的就是 `listReports(100)`。用户历史报表超过 100 张时，看板里的旧报表在索引中缺失 →
`构建血缘图` 会把它们落到「未知数据集」（Task 1 已用单测锁住这个降级路径，图不会崩、节点不会丢）。
这一步把降级变成正解：`GET /reports/{id}` 返回体含 `数据集ID`（`api/routes/reports.py:693-705` 返回 `读取报表` 的全量字段）。

在 `Dashboard.jsx` 的 ④ 之后加：

```jsx
  // 阶段 六十：listReports 最多 100 条，看板里的旧报表可能没进列表项 →
  // 关系视图首次打开时按需补齐（通常 0 次请求，最多看板报表数条）
  const 已补齐 = useRef(new Set());
  useEffect(() => {
    if (视图 !== 'lineage' || !reports.length) return;
    const 缺失 = reports.map((r) => r.报表ID).filter((id) => !报表索引.has(id) && !已补齐.current.has(id));
    if (缺失.length === 0) return;
    缺失.forEach((id) => 已补齐.current.add(id));
    Promise.all(缺失.map((id) => getReport(id).catch(() => null))).then((详情们) => {
      const 可用 = 详情们.filter(Boolean).map((d) => ({
        报表ID: d.报表ID, 数据集ID: d.数据集ID, 标题: d.标题, 图表类型: d.图表类型, 创建时间: d.创建时间,
      }));
      if (可用.length === 0) return;
      setAllReports((prev) => [...prev, ...可用]);
    });
  }, [视图, reports, 报表索引]);
```
