// electron/核心/端口探测.js
const net = require('node:net');

/**
 * 从 startPort 起找第一个可监听的 TCP 端口（探测后立即释放，交给 uvicorn 绑定）。
 * 最多尝试 50 个端口，全部占用抛错。
 */
function 探测空闲端口(startPort = 8000) {
  return new Promise((resolve, reject) => {
    let current = startPort;
    const 尝试 = () => {
      if (current > startPort + 50) {
        return reject(new Error(`端口 ${startPort}-${current - 1} 全部被占用`));
      }
      const srv = net.createServer();
      srv.once('error', () => {
        current += 1;
        尝试();
      });
      srv.once('listening', () => {
        const port = current;
        srv.close(() => resolve(port));
      });
      srv.listen(current, '127.0.0.1');
    };
    尝试();
  });
}

module.exports = { 探测空闲端口 };

// electron/核心/设置存取.js
const fs = require('node:fs');
const path = require('node:path');

/**
 * 读 userData/settings.json；不存在/损坏返回 {}。
 * @param {string} 目录 userData 目录
 */
function 读设置(目录) {
  const file = path.join(目录, 'settings.json');
  try {
    return JSON.parse(fs.readFileSync(file, 'utf-8'));
  } catch {
    return {};
  }
}

/**
 * 合并写入 userData/settings.json（原子写：先写临时文件再 rename）。
 * @param {string} 目录 userData 目录
 * @param {object} partial 要合并的字段
 */
function 写设置(目录, partial) {
  const file = path.join(目录, 'settings.json');
  const next = { ...读设置(目录), ...partial };
  fs.mkdirSync(目录, { recursive: true });
  const tmp = file + '.tmp';
  fs.writeFileSync(tmp, JSON.stringify(next, null, 2), 'utf-8');
  fs.renameSync(tmp, file);
  return next;
}

module.exports = { 读设置, 写设置 };
```

// electron/核心/版本比对.js
/**
 * 语义化版本字符串比较：'1.2.3' vs '1.10.0' 按数字段比较。
 * 返回：'最新'（本地>=最新）| '可更新'（本地<最新）| '未知'（任一解析失败）
 * @param {string} 本地
 * @param {string} 最新
 */
function 比对版本(本地, 最新) {
  const 解析 = (v) => String(v || '').trim().split('.').map(n => parseInt(n, 10));
  const a = 解析(本地);
  const b = 解析(最新);
  if (a.some(Number.isNaN) || b.some(Number.isNaN) || a.length === 0 || b.length === 0) {
    return '未知';
  }
  for (let i = 0; i < Math.max(a.length, b.length); i++) {
    const x = a[i] || 0;
    const y = b[i] || 0;
    if (x > y) return '最新';
    if (x < y) return '可更新';
  }
  return '最新';
}

module.exports = { 比对版本 };
```

const { app, BrowserWindow, ipcMain, shell, dialog } = require('electron');
const path = require('node:path');
const fs = require('node:fs');
const { spawn } = require('node:child_process');
const { 探测空闲端口 } = require('./核心/端口探测');
const { 读设置, 写设置 } = require('./核心/设置存取');
const { 比对版本 } = require('./核心/版本比对');

const APP_NAME = '自助式数据分析Agent平台';
const 数据根目录 = () => path.join(app.getPath('appData'), '数据分析Agent');

let 主窗口 = null;
let 后端进程 = null;

/** 找到打包进 extraResources 的后端可执行文件；开发模式（electron .）用 uv run 启动。 */
function 后端入口() {
  const packaged = path.join(process.resourcesPath, 'backend', 'daa-backend.exe');
  return fs.existsSync(packaged) ? packaged : null;
}

function spawnBackend(port) {
  const 数据目录 = 数据根目录();
  const dataDir = path.join(数据目录, 'data');
  fs.mkdirSync(dataDir, { recursive: true });
  const env = {
    ...process.env,
    AUTH_ENABLED: 'false',
    HOST: '127.0.0.1',
    API_PORT: String(port),
    PORT: String(port),
    DAA_SQLITE_PATH: path.join(dataDir, 'daa.db'),
    DB_BACKEND: 'sqlite',
  };
  const exe = 后端入口();
  if (!exe) {
    throw new Error('未找到后端可执行文件，请先执行 npm run build:backend');
  }
  const proc = spawn(exe, [], {
    env,
    cwd: dataDir,
    windowsHide: true,
    stdio: ['ignore', 'pipe', 'pipe'],
  });
  // 日志落 %APPDATA%\数据分析Agent\logs\backend.log
  const logDir = path.join(数据目录, 'logs');
  fs.mkdirSync(logDir, { recursive: true });
  const logStream = fs.createWriteStream(path.join(logDir, 'backend.log'), { flags: 'a' });
  proc.stdout.pipe(logStream);
  proc.stderr.pipe(logStream);
  proc.on('exit', (code) => {
    if (后端进程 === proc) 后端进程 = null;
    console.log(`后端进程退出 code=${code}`);
  });
  return proc;
}

async function 等后端就绪(port, 超时Ms = 30000) {
  const { execFile } = require('node:child_process');
  const 开始 = Date.now();
  while (Date.now() - 开始 < 超时Ms) {
    try {
      const { data } = await new Promise((resolve, reject) => {
        execFile('curl.exe', ['-s', '-o', 'NUL', '-w', '%{http_code}', `http://127.0.0.1:${port}/healthz`],
          (err, stdout) => err ? reject(err) : resolve({ data: stdout }));
      });
      if (data === '200') return true;
    } catch { /* 未就绪，继续轮询 */ }
    await new Promise(r => setTimeout(r, 300));
  }
  return false;
}

async function 创建主窗口() {
  主窗口 = new BrowserWindow({
    width: 1440, height: 900, minWidth: 1024, minHeight: 700,
    icon: path.join(__dirname, '..', 'logo.ico'),
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      contextIsolation: true,
      nodeIntegration: false,
      sandbox: true,
    },
  });
  主窗口.setMenuBarVisibility(false);
  // 外链交系统浏览器，应用窗口不导航走
  主窗口.webContents.setWindowOpenHandler(({ url }) => {
    if (/^https?:/i.test(url)) shell.openExternal(url);
    return { action: 'deny' };
  });
  主窗口.webContents.on('will-navigate', (e, url) => {
    if (!url.startsWith(`http://127.0.0.1:${全局端口}`)) e.preventDefault();
  });
  const { 读设置 } = require('./核心/设置存取');
  const 设置 = 读设置(app.getPath('userData'));
  if (设置.窗口最大化) 主窗口.maximize();
  主窗口.loadURL(`http://127.0.0.1:${全局端口}`);
}

// 全局保存本次启动端口（will-navigate 校验用）
let 全局端口 = 8000;

// IPC：拖拽取真实路径（webUtils 只能在主进程用）
const { webUtils } = require('electron');
ipcMain.handle('desktop:get-file-path', (event, file) => {
  try { return webUtils.getPathForFile(file); } catch { return ''; }
});
ipcMain.handle('desktop:get-version', () => app.getVersion());
ipcMain.handle('desktop:check-update', async () => {
  // 一期：GitHub Releases API（仓库远端地址与主仓库一致；发布时用同一个 tag）
  try {
    const 响应 = await fetch('https://api.github.com/repos/yundai1009/data-agent-visual-analysis/releases/latest');
    if (!响应.ok) return { 状态: '未知' };
    const 发布 = await 响应.json();
    const 结论 = 比对版本(app.getVersion(), 发布.tag_name.replace(/^v/i, ''));
    return { 状态: 结论, 最新版本: 发布.tag_name, 下载页: 发布.html_url };
  } catch {
    return { 状态: '未知' };
  }
});

// 单实例锁
const gotLock = app.requestSingleInstanceLock();
if (!gotLock) {
  app.quit();
} else {
  app.on('second-instance', () => {
    if (主窗口) { if (主窗口.isMinimized()) 主窗口.restore(); 主窗口.focus(); }
  });

  app.whenReady().then(async () => {
    try {
      全局端口 = await 探测空闲端口(8000);
      后端进程 = spawnBackend(全局端口);
      const ok = await 等后端就绪(全局端口);
      if (!ok) {
        dialog.showErrorBox(`${APP_NAME}`, '后端启动超时，请查看日志：%APPDATA%\\数据分析Agent\\logs\\backend.log');
        app.quit();
        return;
      }
      await 创建主窗口();
      // 托盘在 Task 3 加入
    } catch (e) {
      dialog.showErrorBox(`${APP_NAME}`, `启动失败：${e.message}`);
      app.quit();
    }
  });
}

app.on('before-quit', () => {
  if (后端进程) { 后端进程.kill(); 后端进程 = null; }
});
app.on('window-all-closed', () => {
  // 关窗不退出（托盘常驻在 Task 3 启用；在此之前窗口全关则退出，避免开发期孤儿进程）
  if (!process.env.DAA_TRAY) app.quit();
});

# electron/build-backend/backend_entry.py
# -*- coding: utf-8 -*-
"""桌面版后端入口：从 env 读端口/数据路径，启动 FastAPI 应用。

打包后由 Electron 主进程 spawn 调用；除 uvicorn 标准参数外，
host/port 全部来自环境变量（AUTH_ENABLED / HOST / API_PORT / DAA_SQLITE_PATH）。
"""
import os
import sys
from pathlib import Path


def _项目根() -> Path:
    """打包后项目根 = exe 同级的 _internal 目录的上级（onedir 布局）。"""
    if getattr(sys, "frozen", False):
        # dist-backend/daa-backend.exe → 项目根即 exe 所在目录
        return Path(sys.executable).resolve().parent
    # 开发模式：electron/build-backend/backend_entry.py → 项目根
    return Path(__file__).resolve().parent.parent.parent


def main() -> None:
    # 让 api/config/repositories/后端_核心 可导入
    根 = _项目根()
    if str(根) not in sys.path:
        sys.path.insert(0, str(根))

    host = os.getenv("HOST", "127.0.0.1")
    port = int(os.getenv("API_PORT") or os.getenv("PORT") or "8000")

    import uvicorn
    uvicorn.run(
        "api.main:app",
        host=host,
        port=port,
        log_level=os.getenv("LOG_LEVEL", "warning"),
        # SQLite 单文件库：写锁是进程内 threading.Lock，必须单 worker
        workers=1,
    )

/** 允许拖拽上传的扩展名（与后端 upload 白名单一致：csv/xlsx/xls 等）。 */
const 允许扩展名 = ['.csv', '.xlsx', '.xls'];

/**
 * 从拖拽 File 列表返回可上传文件信息。
 * @param {Array<{name: string, path?: string}>} files
 * @returns {Array<{name: string, path: string, 扩展名: string}>} 可上传项；不可识别项忽略

function 提取可上传文件(files) {

}
